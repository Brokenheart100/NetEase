 private SongDto MapToDto(Song song)
        {
            if (song == null) return null;

            var baseUrl = _configuration["ApiGatewayUrl"];
            Debug.WriteLine($"FilePath {baseUrl}{song.FilePath}");
            return new SongDto
            {
                Id = song.Id,
                Title = song.Title,
                Duration = song.Duration,
                AlbumTitle = song.Album?.Title,
                ArtistName = song.Album?.Artist?.Name,
                CoverImageUrl = !string.IsNullOrEmpty(song.Album?.CoverImageUrl)
                    ? $"{baseUrl}/media/{song.Album.CoverImageUrl}"
                    : null,

                FilePath = !string.IsNullOrEmpty(song.FilePath)
                    ? $"{baseUrl}/media/{song.FilePath}"
                    : null
            };
        }
